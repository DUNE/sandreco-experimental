add_library(sand_debug_debug_streamer)

target_sources(sand_debug_debug_streamer PRIVATE
    debug_streamer.hpp
    debug_streamer.cpp)

target_include_directories(sand_debug_debug_streamer PRIVATE . ${CMAKE_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/src/data/common/ /usr/local/include/EDepSim)

target_link_libraries(sand_debug_debug_streamer PRIVATE ROOT::RIO ROOT::Tree ufw::ufw)

include(GNUInstallDirs)

# The streamer relies on dictionary generation and its own definitions for correct representation of user types.
file(WRITE debug_streamer_types.hpp "/* This file is auto generated by cmake: DO NOT EDIT */\n#include <debug/debug_data.hpp>\n")

file(WRITE debug_streamer_linkdef.h "#ifdef __CLING__\n#pragma link off all globals;\n#pragma link off all classes;\n#pragma link off all functions;\n#pragma link C++ nestedclasses;\n#pragma link C++ nestedtypedefs;\n#pragma link C++ class sand::debug::debug_data+;\n#endif\n")

ROOT_GENERATE_DICTIONARY(sand_debug_debug_streamer_dictionaries debug_streamer_types.hpp MODULE sand_debug_debug_streamer LINKDEF debug_streamer_linkdef.h OPTIONS "-I${CMAKE_SOURCE_DIR}/include")

install(TARGETS sand_debug_debug_streamer EXPORT sandrecoTargets DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libsand_debug_debug_streamer_rdict.pcm
        ${CMAKE_CURRENT_BINARY_DIR}/libsand_debug_debug_streamer.rootmap DESTINATION ${CMAKE_INSTALL_LIBDIR})
