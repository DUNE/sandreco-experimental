#The streamer relies on dictionary generation and its own definitions for
#correct representation of any user type. This requires all said types to be known.
#The code below attempts to grab all user types from include/
file(GLOB_RECURSE USER_TYPES
     RELATIVE "${CMAKE_SOURCE_DIR}/include"
     "${CMAKE_SOURCE_DIR}/include/*.hxx"
     "${CMAKE_SOURCE_DIR}/include/*.hpp"
     "${CMAKE_SOURCE_DIR}/include/*.h")

set(USER_TYPES_INCLUDES ${USER_TYPES})
set(COMMENT_NO_EDIT "/* This file is auto generated by cmake: DO NOT EDIT */\n")

#Generate the include file tree_streamer_types.hpp
list(TRANSFORM USER_TYPES_INCLUDES PREPEND "#include <")
list(TRANSFORM USER_TYPES_INCLUDES APPEND ">\n")

set(STREAMER_TYPES "${CMAKE_CURRENT_BINARY_DIR}/tree_streamer_types.hpp")
set(STREAMER_LINKDEF "${CMAKE_CURRENT_BINARY_DIR}/tree_streamer_linkdef.h")


file(WRITE ${STREAMER_TYPES} ${COMMENT_NO_EDIT})
file(APPEND ${STREAMER_TYPES} "#ifndef UFW_IMPLEMENT_STREAMER_FOR_TYPE\n#define UFW_IMPLEMENT_STREAMER_FOR_TYPE(T)\n#endif\n")
file(APPEND ${STREAMER_TYPES} "#include <data.h>\n") # Which in turn includes ufw/data.hpp
file(APPEND ${STREAMER_TYPES} ${USER_TYPES_INCLUDES})

#Generate the linkdef file tree_streamer_linkdef.h
set(USER_TYPES_LINKDEF "")
foreach(USER_TYPE_PATH ${USER_TYPES})
  file(STRINGS "${CMAKE_SOURCE_DIR}/include/${USER_TYPE_PATH}" USER_FILE_LINES REGEX "^UFW_DECLARE_(MANAGED|UNMANAGED)_DATA")
  foreach(LINE ${USER_FILE_LINES})
    string(REGEX MATCH "\\(([^()]*)\\)" _ ${LINE})
    list(APPEND USER_TYPES_LINKDEF ${CMAKE_MATCH_1})
  endforeach()
endforeach()

list(TRANSFORM USER_TYPES_LINKDEF PREPEND "#pragma link C++ class ")
list(TRANSFORM USER_TYPES_LINKDEF APPEND "+\;\n")

file(WRITE ${STREAMER_LINKDEF} ${COMMENT_NO_EDIT})
file(APPEND ${STREAMER_LINKDEF} "#ifdef __CLING__\n#pragma link off all globals;\n#pragma link off all classes;\n#pragma link off all functions;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ nestedclasses;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ nestedtypedefs;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::context_id+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::data::data_base+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::data::managed_tag+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::data::instanced_tag+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::data::context_tag+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class ufw::data::base<ufw::data::managed_tag,ufw::data::instanced_tag,ufw::data::context_tag>+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::truth_index+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::truth<sand::truth_index>+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::truth<sand::ecal::pes_container::photo_electron>+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::truth<sand::grain::hits::photon>+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::geo_id+;\n")
file(APPEND ${STREAMER_LINKDEF} "#pragma link C++ class sand::channel_id+;\n")
file(APPEND ${STREAMER_LINKDEF} ${USER_TYPES_LINKDEF})
file(APPEND ${STREAMER_LINKDEF} "#endif\n")

include_directories(${duneanaobj_INCLUDE_DIR})

ROOT_GENERATE_DICTIONARY(sand_root_tree_streamer_dictionaries ${STREAMER_TYPES} MODULE sand_root_tree_streamer LINKDEF ${STREAMER_LINKDEF})

install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libsand_root_tree_streamer_rdict.pcm
        ${CMAKE_CURRENT_BINARY_DIR}/libsand_root_tree_streamer.rootmap DESTINATION ${CMAKE_INSTALL_LIBDIR})
